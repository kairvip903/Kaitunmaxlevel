local function new(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props) do o[k] = v end
	o.Parent = parent
	return o

local function corner(o, r) return new("UICorner", {CornerRadius = r and UDim.new(0, r) or UDim.new(1, 0)}, o) end

local function label(parent, props)
	local base = {BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, TextSize = 11, TextColor3 = TEXT,

local function sweep(o)
	local st = new("UIStroke", {Color = Color3.new(1, 1, 1), Thickness = 1.4, Transparency = 0, ApplyStrokeMode = Enum.ApplyStrokeMode.Border}, o)
	local g = new("UIGradient", {Transparency = NumberSequence.new({NumberSequenceKeypoint.new(0, 0.72), NumberSequenceKeypoint.new(0.4, 0.72),

	TweenService:Create(g, TweenInfo.new(7, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1), {Rotation = 360}):Play()
	return st
end
local function short(n)
	n = tonumber(n) or 0
	if n >= 1e9 then return string.format("%.2fB", n / 1e9) end
	if n >= 1e6 then return string.format("%.2fM", n / 1e6) end
	if n >= 1e3 then return string.format("%.1fK", n / 1e3) end
	return tostring(math.floor(n))
end

local function digits(v) return (tostring(v):gsub("%D", "")) end
local function plain(v) return (tostring(v):gsub("<[^>]*>", "")) end
local function hudIcon(value, hints)
	local want = digits(math.floor(tonumber(value) or -1))
	if want == "" then return nil end
	local pg = LP:FindFirstChildOfClass("PlayerGui")
	local cands, fallback = {}, {}
	for _, o in ipairs(pg and pg:GetDescendants() or {}) do
		if o:IsA("TextLabel") and o.Visible and not o:IsDescendantOf(gui) and digits(plain(o.Text)) == want then
			local key = (o:GetFullName() .. " " .. o.Text):lower()
			local hit = false
			for _, h in ipairs(hints or {}) do
				if key:find(h, 1, true) then hit = true end
			end
			table.insert(hit and cands or fallback, o)
		end
	end
	if #want >= 3 then
		for _, o in ipairs(fallback) do cands[#cands + 1] = o end
	end
	for _, o in ipairs(cands) do
		do
			for _, sib in ipairs(o.Parent:GetChildren()) do
				if sib ~= o and (sib:IsA("ImageLabel") or sib:IsA("ImageButton")) and sib.Visible and sib.Image ~= "" then return sib, o end
			end
			for _, ch in ipairs(o:GetChildren()) do
				if ch:IsA("ImageLabel") and ch.Image ~= "" then return ch, o end
			end
			for _, sib in ipairs(o.Parent:GetChildren()) do
				if sib ~= o and sib:IsA("TextLabel") and sib.Visible and plain(sib.Text) ~= "" and digits(plain(sib.Text)) == ""
				   and #plain(sib.Text) <= 4 then return sib, o end
			end
			local sym = plain(o.Text):match("^[^%d%s,%.]+")
			if sym and sym ~= "" then
				local c = o:Clone()
				c.RichText = false
				c.Text = sym
				local hex = o.Text:match("color%s*=%s*[\"'](#%x%x%x%x%x%x)")
				local r, g, b = o.Text:match("rgb%((%d+)%s*,%s*(%d+)%s*,%s*(%d+)%)")
				if hex then
					c.TextColor3 = Color3.fromRGB(tonumber(hex:sub(2, 3), 16), tonumber(hex:sub(4, 5), 16), tonumber(hex:sub(6, 7), 16))
				elseif r then
					c.TextColor3 = Color3.fromRGB(tonumber(r), tonumber(g), tonumber(b))
				end
				return c, o
			end
			return nil, o
		end
	end
end
local function placeIcon(k, src, num)
	local t = STATICON[k]
	if t and num then
		pcall(function()
			t.val.TextColor3 = num.TextColor3
			local g = num:FindFirstChildOfClass("UIGradient")
			if g and not t.val:FindFirstChildOfClass("UIGradient") then g:Clone().Parent = t.val end
		end)
	end
	if not (t and src) then return false end
	local c = src.Parent and src:Clone() or src
	for _, d in ipairs(c:GetDescendants()) do
		if d:IsA("LuaSourceContainer") then d:Destroy() end
	end
	c.AnchorPoint = Vector2.new(0.5, 0.5)
	c.Position = UDim2.fromOffset(20, 19)
	c.Size = UDim2.fromOffset(24, 24)
	c.Rotation = 0
	c.Visible = true
	if c:IsA("TextLabel") then
		c.TextScaled = true
		c.BackgroundTransparency = 1
	end
	pcall(function() c.ZIndex = 2 end)
	c.Parent = t.img.Parent
	t.img.Visible, t.bar.Visible = false, false
	t.cap.Position = UDim2.fromOffset(38, 5)
	t.val.Position = UDim2.fromOffset(38, 15)
	t.val.Size = UDim2.new(1, -44, 0, 16)
	return true
end

local function rarOf(it, kind)
	if it.Race then return it.Rarity end
	local r = math.max(tonumber(it.Rarity) or 0, kind == "Melee" and MELEE_RAR[it.Name] or 0, RAR_NAME[ALIAS[it.Name] or it.Name] or RAR_NAME[it.Name] or 0)
	if kind == "Blox Fruit" then
		local f = FRUIT_RANK[tostring(it.Name):match("^([^%-]+)"):lower()]
		if f and f > r then r = f end
	end
	return r
end
local function rarityByName(name, kind)
	if kind == "Melee" and MELEE_RAR[name] then return rarOf({Name = name}, "Melee") end
	for _, it in ipairs(K.Inv) do
		if it.Name == name then return rarOf(it, it.Type) end
	end
	return RAR_NAME[name]
end
local function rarGrad(rc) return ColorSequence.new(rc:Lerp(Color3.new(0, 0, 0), 0.15), rc:Lerp(Color3.new(0, 0, 0), 0.72)) end
local function fadeAll(f, info, show)
	local list = f:GetDescendants()
	list[#list + 1] = f
	for _, o in ipairs(list) do
		local k = o:IsA("UIStroke") and "Transparency" or o:IsA("TextLabel") and "TextTransparency" or o:IsA("ImageLabel") and "ImageTransparency"
		for _, key in ipairs({k, o:IsA("GuiObject") and "BackgroundTransparency" or nil}) do
			local v = key and o[key]
			if v and v < 1 then
				if show then o[key] = 1 end
				TweenService:Create(o, info, {[key] = show and v or 1}):Play()
			end
		end
	end
end
local function popIn(f, delay)
	local info = TweenInfo.new(0.3, Enum.EasingStyle.Quint, Enum.EasingDirection.Out, 0, false, delay)
	local goal = f.Position
	f.Position = goal + UDim2.fromOffset(0, 18)
	TweenService:Create(f, info, {Position = goal}):Play()
	fadeAll(f, info, true)
end
local function popOut(f, delay)
	f.Name = "Out"
	local info = TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.In, 0, false, delay)
	TweenService:Create(f, info, {Position = f.Position - UDim2.fromOffset(0, 14)}):Play()
	fadeAll(f, info, false)
	task.delay(delay + 0.18, function() f:Destroy() end)
end

local lastSig, lastTab = "", nil
rebuild = function(force)
	local groups = {}
	for _, t in ipairs(TABS) do groups[t[1]] = {} end
	for name, mas in pairs(K.Melee) do table.insert(groups.Melee, {Name = name, Mastery = mas}) end
	for _, it in ipairs(K.Inv) do
		if groups[it.Type] then
			table.insert(groups[it.Type], it)
		end
		if it.Type ~= "Title" and it.Type ~= "Race" and it.Type ~= "Item" then
			table.insert(groups.Item, it)
		end
	end
	local race = "Human"
	pcall(function() race = tostring(LP.Data.Race.Value) end)
	local sym = RACE_SYMBOL[race] and SPRITE_CAT["All/" .. RACE_SYMBOL[race] .. " Symbol"]
	local function raceKey(n)
		local b = tostring(n):gsub("%s*[Vv]%d$", "")
		return RACE_SYMBOL[b] or b
	end
	K.RaceIcon = sym
	for _, it in ipairs(K.Inv) do
		if it.Type == "Race" and it.Icon and raceKey(it.Name) == raceKey(race) then
			K.RaceIcon = it.Icon
			break
		end
	end
	local total, sig = 0, {K.Tab, tostring(K.RaceV), race}
	for _, t in ipairs(TABS) do
		local g = groups[t[1]]
		do
			table.sort(g, function(a, b)
				local ra, rb = rarOf(a, t[1]), rarOf(b, t[1])
				if ra ~= rb then return ra > rb end
				local ma, mb = tonumber(a.Mastery) or 0, tonumber(b.Mastery) or 0
				if ma ~= mb then return ma > mb end
				return (tonumber(a.Count) or 0) > (tonumber(b.Count) or 0)
			end)
			for _, it in ipairs(g) do
				if t[1] ~= "Item" or it.Type == "Item" or not groups[it.Type] then total = total + 1 end
			end
		end
		local sel = K.Tab == t[1]
		TABBTN[t[1]].Text = t[2] .. "  " .. #g
		local tc = TAB_COL[t[1]] or ACCENT
		TABBTN[t[1]].BackgroundColor3 = tc
		TweenService:Create(TABBTN[t[1]], TweenInfo.new(0.22, Enum.EasingStyle.Quint), {BackgroundTransparency = sel and 1 or 0.82,
			TextColor3 = sel and Color3.new(1, 1, 1) or tc:Lerp(Color3.new(1, 1, 1), 0.35)}):Play()
		if sel and pillIdx ~= t[1] then
			local first = pillIdx == nil
			pillIdx = t[1]
			local goal = {Position = UDim2.new((TABBTN[t[1]].Position.X.Scale), 3, 0, 0), BackgroundColor3 = tc}
			if first then
				pill.Position, pill.BackgroundColor3 = goal.Position, goal.BackgroundColor3
			else
				TweenService:Create(pill, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), goal):Play()
			end
		end
		for _, it in ipairs(g) do sig[#sig + 1] = it.Name .. ":" .. tostring(it.Mastery) end
	end
	countL.Text = "[" .. total .. "]"
	sig = table.concat(sig, "|")
	if sig == lastSig and not force then return end
	lastSig = sig
	local anim, outN = K.Tab ~= lastTab and 0 or nil, 0
	lastTab = K.Tab
	for _, c in ipairs(grid:GetChildren()) do
		if c:IsA("Frame") and c.Name ~= "Out" then
			if anim and outN < 9 then
				popOut(c, outN * 0.015)
				outN = outN + 1
			else
				c:Destroy()
			end
		end
	end
	if anim and outN > 0 then anim = 0.14 + outN * 0.015 end
	local rows = -1
	for i, it in ipairs(groups[K.Tab]) do rows = tile(i, it, K.Tab, anim) end
	grid.CanvasSize = UDim2.fromOffset(0, (rows + 1) * 50 + 6)
	if anim then grid.CanvasPosition = Vector2.new() end
end

local itemSvc, itemCfg, catsSeen, keysLogged = nil, nil, {}, false
local function rarValue(v)
	local t = typeof(v)
	if t == "number" then return v end
	if t == "EnumItem" then v = v.Name end
	if type(v) == "table" then v = v.Name or v.Value or v.Rarity end
	if type(v) == "number" then return v end
	if type(v) == "string" then
		local lv = v:lower()
		for _, name in ipairs({"uncommon", "common", "rare", "legendary", "mythical", "premium"}) do
			if lv == name then return RARITY_NUM[name:sub(1, 1):upper() .. name:sub(2)] end
		end
		for _, name in ipairs({"uncommon", "common", "legendary", "mythical", "premium", "rare"}) do
			if lv:find(name, 1, true) then return RARITY_NUM[name:sub(1, 1):upper() .. name:sub(2)] end
		end
		return tonumber(v)
	end
end
local function findRarity(c)
	local seen, queue = {}, {{c, 0}}
	while #queue > 0 do
		local node = table.remove(queue, 1)
		local t, d = node[1], node[2]
		if type(t) == "table" and not seen[t] then
			seen[t] = true
			for k, v in pairs(t) do
				local lk = tostring(k):lower()
				if lk:find("rarity", 1, true) or lk == "tier" then
					local n = rarValue(v)
					if n then return math.clamp(math.floor(n), 0, 5) end
				end
				if type(v) == "table" and d < 3 then queue[#queue + 1] = {v, d + 1} end
			end
		end
	end
end

	local out, melee = {}, {}
	pcall(function()

		itemSvc = itemSvc or require(RS:WaitForChild("ItemReplicationService", 5))
		itemCfg = itemCfg or require(RS:WaitForChild("ItemConfig", 5))

						catsSeen[cat] = true
						local rar = findRarity(c) or 0

							keysLogged = true
							local ks = {}
							for k2, v2 in pairs(c) do ks[#ks + 1] = tostring(k2) .. "=" .. (type(v2) == "table" and "{}" or tostring(v2)) end
							print("[Lucious Kaitun] ItemConfig fields (" .. tostring(nm) .. "): " .. table.concat(ks, ", "))
						end
						local sp = c.Display.Sprite
						local icon = type(sp) == "table" and type(sp.Image) == "string"
							and {img = sp.Image, off = sp.ImageRectOffset, size = sp.ImageRectSize} or nil
						if cat == "Melee" or cat:find("Fight") then
							melee[tostring(nm)] = tonumber(ms) or 0
							MELEE_RAR[tostring(nm)] = rar
							if icon then MELEE_SPRITE[tostring(nm)] = icon end
						else
							out[#out + 1] = {Name = tostring(nm), Type = cat, Mastery = tonumber(ms), Count = it.Value, Icon = icon, Rarity = rar}
						end
					end
				end
			end
		end
	end)
	if #out == 0 then
		local ok, res = pcall(function()
			return game:GetService("ReplicatedStorage").Remotes.CommF_:InvokeServer("getInventory")
		end)
		if ok and type(res) == "table" then
			for _, it in pairs(res) do
				if type(it) == "table" and it.Name then
					out[#out + 1] = {Name = tostring(it.Name), Type = tostring(it.Type), Mastery = it.Mastery, Count = it.Count}
				end
			end
		end
	end
	if #out > 0 then K.Inv = out end
	for name, m in pairs(melee) do
		if (K.Melee[name] or -1) < m then K.Melee[name] = m end
	end
end

end

local function paintTop()
	local all = {}
	for name, mas in pairs(K.Melee) do all[#all + 1] = {Name = name, Type = "Melee", Mastery = mas} end
	for _, it in ipairs(K.Inv) do
		if it.Type == "Sword" or it.Type == "Gun" or it.Type == "Blox Fruit" then all[#all + 1] = it end
	end
	table.sort(all, function(a, b) return (tonumber(a.Mastery) or 0) > (tonumber(b.Mastery) or 0) end)
	for i, r in ipairs(TOPW) do
		local it = all[i]
		r.dot.BackgroundColor3 = it and TYPE_COL[it.Type] or SUB
		local rc = it and RARITY_COL[rarOf(it, it.Type)]
		r.bg.BackgroundTransparency = rc and 0.2 or 1
		if rc then r.grad.Color = rarGrad(rc) end
		local ic = it and iconOf(it.Name, it.Type, it.Icon)
		if ic then setIcon(r.ico, ic) end
		r.ico.Visible = ic ~= nil
		r.dot.Visible = ic == nil
		r.name.Text = it and ((it.Type == "Blox Fruit" and "Fruit" or it.Type) .. "  /  " .. it.Name) or "Empty slot"
		r.name.TextColor3 = it and TEXT or SUB
		r.mas.Text = it and it.Mastery and ("Mas " .. tostring(it.Mastery)) or ""
		r.mas.TextColor3 = it and tonumber(it.Mastery) and masColor(tonumber(it.Mastery)) or SUB
	end
end

	local function twSpd(base)
		local c = TW.cap

			return base
		end
		if os.clock() >= TW.raiseAt then

		end

	end
	local function rawTween(pos)
		local _, hrp, hum = parts()
		if not hrp or (hrp.Position - pos).Magnitude <= 6 then return end
		hum.AutoRotate = false
		local att = Instance.new("Attachment")
		att.Parent = hrp
		local lv = Instance.new("LinearVelocity")
		lv.Attachment0, lv.MaxForce, lv.VectorVelocity = att, math.huge, Vector3.zero
		lv.RelativeTo = Enum.ActuatorRelativeTo.World
		lv.Parent = hrp
		local rot = hrp.CFrame.Rotation
		local lastSet, tickAt, seg, rest = nil, os.clock(), 0, 0
		local t0, limit = os.clock(), (hrp.Position - pos).Magnitude / 150 * 3 + 30
		while alive and hrp.Parent and hum.Health > 0 and os.clock() - t0 < limit do
			local here = hrp.Position
			if (pos - here).Magnitude <= 6 then break end
			local now = os.clock()
			local dt = math.min(now - tickAt, 0.1)
			tickAt = now
			hrp.AssemblyLinearVelocity = Vector3.zero
			if hrp.Anchored then
				lastSet = nil
			else
				if lastSet and (here - lastSet).Magnitude > 8 then
					TW.cap, TW.raiseAt = math.max((TW.cap or SPEED) * 0.7, 60), now + 3
				end
				if rest > 0 then
					rest = rest - 1
					lastSet = here
				else
					local d = pos - here
					local step = twSpd(SPEED) * dt
					local np = d.Magnitude <= step and pos or here + d.Unit * step
					seg = seg + (np - here).Magnitude
					if seg >= 100 then seg, rest = 0, 1 end
					hrp.CFrame = CFrame.new(np) * rot
					lastSet = np
				end
			end
			hrp.AssemblyAngularVelocity = Vector3.zero
			RunService.Heartbeat:Wait()
		end
		pcall(function() lv:Destroy() end)
		pcall(function() att:Destroy() end)
		if hrp.Parent then hrp.AssemblyLinearVelocity = Vector3.zero end
	end

	local function strike(target, mobs)
		local _, hrp = parts()
		if not hrp then return end
		local tp = target.hrp.Position
		if not gather or (Vector3.new(tp.X, gather.Y, tp.Z) - gather).Magnitude > 80 then gather = tp end
		local spot = gather + Vector3.new(0, HOVER, 0)
		if (hrp.Position - spot).Magnitude > 150 then
			travel(spot)
			return
		end
		if (hrp.Position - spot).Magnitude > 0.5 then hrp.CFrame = CFrame.new(spot) * hrp.CFrame.Rotation end
		local others = {}
		pinned = {}
		for _, e in ipairs(mobs) do
			if (e.hrp.Position - gather).Magnitude < 1200 then
				pinned[#pinned + 1] = e
				pcall(function()
					if owns(e.hrp) then
						e.hrp.CanCollide = false
						e.hrp.AssemblyLinearVelocity = Vector3.zero
						e.hrp.CFrame = CFrame.new(gather) * e.hrp.CFrame.Rotation
					end
				end)
				if e ~= target then others[#others + 1] = {e.model, hitPart(e.model, e.hrp)} end
			end
		end
		if RA and RH then
			local hp = hitPart(target.model, target.hrp)
			pcall(RA.FireServer, RA, 0)
			pcall(RH.FireServer, RH, hp, others)
			atkPay = {hp, others, target.hum, os.clock()}
		end
	end

			if not target or (e.hrp.Position - hrp.Position).Magnitude < (target.hrp.Position - hrp.Position).Magnitude then target = e end
		end
		return target
	end

	local function nearestChest()
		local _, hrp = parts()
		if not hrp then return end
		local best, bp, bd
		local function consider(inst)
			local okA, dis = pcall(function() return inst:GetAttribute("IsDisabled") end)
			if okA and dis then return end
			local p = inst:IsA("BasePart") and inst or (inst:IsA("Model") and (inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart", true)))
			if not p then return end
			for _, q in ipairs(LEV.minkSkip) do

			local d = (p.Position - hrp.Position).Magnitude
			if not bd or d < bd then best, bp, bd = inst, p, d end
		end
		pcall(function()
			for _, inst in ipairs(game:GetService("CollectionService"):GetTagged("_ChestTagged")) do consider(inst) end
		end)
		return best, bp
	end

	local fruitSkip = {}
	local function groundFruit(hrp)
		local best, bd
		local function consider(o)
			if (fruitSkip[o] or 0) > os.clock() then return end
			local holder = o:FindFirstAncestorWhichIsA("Model")
			if holder and holder:FindFirstChildOfClass("Humanoid") then return end
			local p = o:FindFirstChild("Handle") or o:IsA("Model") and (o.PrimaryPart or o:FindFirstChildWhichIsA("BasePart", true)) or o
			if not (p and p:IsA("BasePart")) or p.Position.Y < SUB_Y then return end
			local d = (p.Position - hrp.Position).Magnitude
			if d < (bd or 8000) then best, bd = {obj = o, part = p}, d end
